6. Python Full Flow

Python 3
import requests, time

API = "https://YOUR_DOMAIN.com"
KEY = "ats_your_key_here"
H = {"X-API-Key": KEY, "Content-Type": "application/json"}

print(requests.get(f"{API}/v1/balance", headers=H).json())

countries = requests.get(f"{API}/v1/countries", params={"server": 2}, headers=H).json()
print(countries["countries"][:3])

buy = requests.post(f"{API}/v1/buy", headers=H, json={
    "server": 2,
    "country": 22
}).json()
print("Phone:", buy.get("phone"))
order_id = buy["order_id"]

for _ in range(30):
    time.sleep(5)
    st = requests.get(f"{API}/v1/order/{order_id}", headers=H).json()
    if st["status"] == "DONE":
        print("OTP:", st["code"])
        break
    print("Status:", st["status"])

🐍 Python Bot Integration (Telethon)

Telethon async library hai, isliye requests ki jagah httpx use karo. requests event loop ko block karta hai, jisse bot sab users ke liye slow ho jaata hai.

⚠️
Install: pip install httpx

1. API Client Wrapper

reseller_api.py
import asyncio
import httpx

API_BASE = "https://YOUR_DOMAIN.com"
API_KEY  = "ats_your_key_here"   # .env mein rakho, code mein hardcode mat karo

ERRORS = {
    400: "Stock nahi ya params galat",
    401: "API key galat hai",
    402: "Reseller balance kam hai",
    403: "API key disabled hai, admin se contact karo",
    404: "Order nahi mila",
    502: "Provider error, thodi der baad try karo",
    503: "Provider configure nahi hai",
}

class ResellerError(Exception):
    def __init__(self, status, msg):
        super().__init__(msg)
        self.status = status

class ResellerAPI:
    def __init__(self):
        # Ek hi client bot start pe banao, har request pe naya mat banao
        self.client = httpx.AsyncClient(
            base_url=API_BASE,
            headers={"X-API-Key": API_KEY},
            timeout=20,
        )

    async def _req(self, method, path, **kw):
        r = await self.client.request(method, path, **kw)
        if r.status_code != 200:
            raise ResellerError(r.status_code, ERRORS.get(r.status_code, r.text))
        return r.json()

    async def balance(self):
        return await self._req("GET", "/v1/balance")

    async def buy(self, **body):
        # buy(server=1, country_code="IN") | buy(server=2, country=22)
        return await self._req("POST", "/v1/buy", json=body)

    async def order(self, order_id):
        return await self._req("GET", f"/v1/order/{order_id}")

    async def cancel(self, order_id):
        return await self._req("POST", f"/v1/cancel/{order_id}")

    async def retry(self, order_id):
        return await self._req("POST", f"/v1/retry/{order_id}")

    async def wait_code(self, order_id, timeout=300, interval=5):
        # Docs ke hisaab se 4-6 sec poll karo
        elapsed = 0
        while elapsed < timeout:
            st = await self.order(order_id)
            if st["status"] == "DONE":
                return st["code"]
            if st["status"] == "CANCELLED":
                raise ResellerError(0, "Order cancel ho gaya")
            await asyncio.sleep(interval)
            elapsed += interval
        raise TimeoutError("OTP time pe nahi aaya")

api = ResellerAPI()

2. Telethon Button Handler

bot.py
from telethon import events
from reseller_api import api, ResellerError

# bot = TelegramClient(...) already bana hua hai

@bot.on(events.CallbackQuery(data=b"buy_s1_IN"))
async def buy_tg_account(event):
    try:
        order = await api.buy(server=1, country_code="IN")
    except ResellerError as e:
        await event.answer(str(e), alert=True)
        return

    oid = order["order_id"]
    await event.edit(f"📱 Number: {order['phone']}\nOTP ka wait karo...")

    try:
        code = await api.wait_code(oid)
        await event.respond(f"✅ Code: {code}")
    except ResellerError as e:
        await event.respond(f"❌ {e}")
    except TimeoutError:
        await event.respond("⏱️ Code time pe nahi aaya. Retry ya cancel karo.")
💡
Bot ke liye tips:
• Order ko apne DB mein save karo (order_id, user_id, status), taaki bot restart ke baad bhi polling resume ho sake.
• cancel sirf WAIT_CODE pe, aur retry sirf WAIT_RETRY pe karo.
• Server 1 pe strictly no refund hai, isliye buy aur cancel pe confirm step lagao.
• Har buy se pehle balance check karna optional hai, lekin 402 ko zaroor handle karo.
• Heavy polling ke liye asyncio.create_task() use karo taaki handler jaldi return ho.
